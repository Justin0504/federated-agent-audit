  \draw[figtick] (0.74,\y) -- (0.78,\y) node[figticklabel, left] {\y};

  \node[figticklabel, align=center, anchor=north] at (\x,-5) {\l};

\draw[figseries=figrose] (1,81.5) -- (2,78) -- (3,46);

egress.}
